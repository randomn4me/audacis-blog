Keep the emitted markup free of blank lines: a component call in markdown is expanded before the markdown pass, so a blank line ends the HTML block and the indented lines after it become a code block. Hence the `{%- -%}` on every tag below. -#} {% component img(path: string, alt: string = "", width: integer = 800, quality: integer = 85, source: string = "", source_label: string = "Source", raw: bool = false) %} {% endcomponent img %} {% component gallery(paths: array, alts: array = [], columns: integer = 3, thumb_width: integer = 300, full_width: integer = 1600, quality: integer = 85) %}
